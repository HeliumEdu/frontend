# Fingerprint rules for the `frontend` Sentry project. Pasted verbatim into
# Settings -> Projects -> frontend -> Issue Grouping -> Fingerprint Rules.

# Syncfusion calendar internals. Upstream, unfixable, splits per call site. Full
# pub-cache path on web/dart2js, bare filename on AOT.
error.value:"*Null check operator used on a null value*" stack.abs_path:"**syncfusion_flutter_calendar**" -> helium-syncfusion-calendar title="Syncfusion calendar null-check (upstream)"
error.value:"*Null check operator used on a null value*" stack.abs_path:"**sfcalendar.dart" -> helium-syncfusion-calendar title="Syncfusion calendar null-check (upstream)"

# Syncfusion schedule-view scroll. Upstream iterates _forwardWidgetHeights.length (a Map)
# while indexing _nextDates[i] (a List).
error.value:"*RangeError*" stack.abs_path:"**syncfusion_flutter_calendar**" -> helium-syncfusion-calendar title="Syncfusion calendar (upstream)"
error.value:"*RangeError*" stack.abs_path:"**sfcalendar.dart" -> helium-syncfusion-calendar title="Syncfusion calendar (upstream)"

# Flutter web CanvasKit context loss. Upstream engine, no app code in the path. The empty
# Field '' is throwUnnamedLateFieldNI; our own late fields are named and cannot match.
error.value:"*Field '' has not been initialized*" -> helium-flutter-web-canvaskit title="Flutter web CanvasKit context loss (upstream)"

# flutter_quill RenderEditableContainerBox.childAtPosition. Quill alone throws this string.
error.value:"*targetChild should not be null*" -> helium-quill-targetchild title="Quill editor targetChild (upstream)"

# API server faults, otherwise split across DioMixin.request / page_header_save_button /
# root /. Status-agnostic: sentry_service.dart ignoreErrors already drops 4xx, timeouts and
# connection errors client-side.
error.value:"*DioException*bad response*" -> helium-dio-bad-response title="API bad response (server)"

# Asset fetch failures, split per filename. Fonts first, first match wins. Wildcards stand
# in for the literal quotes around the asset name.
error.value:"*Unable to load asset:*assets/fonts/*" -> helium-asset-load-font title="Web font asset failed to load"
error.value:"*Unable to load asset:*" -> helium-asset-load-other title="Asset failed to load (non-font)"
